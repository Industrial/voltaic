================================================================
a binding's else tail, inline, diverging with break
================================================================

drained : Vec<u64> -> u64
drained xs =
  mut rest = xs
  mut total = 0
  loop
    Some v = .pop (&mut rest) else break
    total := total + v
  total

================================================================
a binding's else tail, indented, which is the shape guard always had
================================================================

opened : Option<u64> -> u64
opened held =
  Some n = held else
    return 0
  n
