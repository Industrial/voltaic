================================================================
a ladder in value position
================================================================

settled : u64 -> u64
settled n =
  adjusted =
    | n > 5 = n - 5
    | _ = n
  adjusted * 2

================================================================
a ladder's arms are the clause ladder's arms, pattern rows included
================================================================

named : Option<u64> -> &'static str
named held =
  shown =
    | held is Some _n = "some"
    | _ = "none"
  shown

================================================================
ladders nest through a binding, not through an arm
================================================================

graded : u64 -> u64 -> &'static str
graded spend floor =
  inner =
    | floor > 100 = "high"
    | _ = "low"
  outer =
    | spend > 1_000 = inner
    | _ = "small"
  outer
